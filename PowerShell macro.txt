---------------------------------------------------------
SCRIPT PARA OBTENER LISTA DE ARCHIVOS SIN EXTENSION
---------------------------------------------------------

$carpeta = $PSScriptRoot
$archivosPorExtension = Get-ChildItem -Path $carpeta -Recurse -File | Group-Object Extension | ForEach-Object {
    $extension = $_.Name.TrimStart('.')
    $archivos = $_.Group | Select-Object @{Name="Nombre"; Expression={$_.Name -replace '\.\w+$', ''}}
    "`n$extension archivos:"
    $archivos | ForEach-Object { "$($_.Nombre)" }
}

$archivosPorExtension | Out-File -FilePath "ListeDesFichiers.txt"
